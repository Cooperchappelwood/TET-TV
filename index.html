// ═══════════════════════════════════════════════════════════════════ // THE EVENT TEAM — Office Board Google Apps Script // Paste this entire file into your Google Apps Script editor, // then redeploy (Deploy → Manage Deployments → edit → New Version) // ═══════════════════════════════════════════════════════════════════ // // SHEET STRUCTURE (3 tabs): // // Tab: "Events" // Columns: name | date | location | color // // Tab: "Settings" // Columns: key | value // Rows: officeDays | 1,2,3,4,5 // logoUrl | (optional URL) // // Tab: "Closures" // Columns: date | note // // ═══════════════════════════════════════════════════════════════════ // ── READ — called by the board every 5 minutes ── function doGet(e) { try { const data = readAllData(); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // ── WRITE — called when someone saves from the board ── function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // ── Write Events ── if (payload.events && Array.isArray(payload.events)) { const evSheet = ss.getSheetByName('Events'); // Clear all data rows below header const lastRow = evSheet.getLastRow(); if (lastRow > 1) evSheet.getRange(2, 1, lastRow - 1, 4).clearContent(); // Write new rows payload.events.forEach((ev, i) => { evSheet.getRange(i + 2, 1, 1, 4).setValues([[ ev.name || '', ev.date || '', ev.location || '', ev.color || '#D4A843' ]]); }); } // ── Write Settings ── const setSheet = ss.getSheetByName('Settings'); // Update officeDays row if (payload.officeDays !== undefined) { updateSettingsRow(setSheet, 'officeDays', (payload.officeDays || []).join(',')); } if (payload.logoUrl !== undefined) { updateSettingsRow(setSheet, 'logoUrl', payload.logoUrl || ''); } // ── Write Closures ── if (payload.closures && Array.isArray(payload.closures)) { const clSheet = ss.getSheetByName('Closures'); const lastRow = clSheet.getLastRow(); if (lastRow > 1) clSheet.getRange(2, 1, lastRow - 1, 2).clearContent(); payload.closures.forEach((date, i) => { clSheet.getRange(i + 2, 1, 1, 2).setValues([[date, '']]); }); } // Return the fresh data so the board can confirm what was saved const fresh = readAllData(); return ContentService .createTextOutput(JSON.stringify(fresh)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // ── Helper: find a key row in Settings and update its value ── function updateSettingsRow(sheet, key, value) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === key) { sheet.getRange(i + 1, 2).setValue(value); return; } } // Key not found — append it sheet.appendRow([key, value]); } // ── Helper: read all data from the sheet into a JS object ── function readAllData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Events const evSheet = ss.getSheetByName('Events'); const evData = evSheet.getDataRange().getValues(); const events = []; for (let i = 1; i < evData.length; i++) { const [name, date, location, color] = evData[i]; if (!name || !date) continue; let dateStr = ''; if (date instanceof Date) { const y = date.getFullYear(); const m = String(date.getMonth() + 1).padStart(2, '0'); const d = String(date.getDate()).padStart(2, '0'); dateStr = `${y}-${m}-${d}`; } else { dateStr = String(date).trim(); } events.push({ name: String(name).trim(), date: dateStr, location: String(location || '').trim(), color: String(color || '#D4A843').trim() }); } // Settings const setSheet = ss.getSheetByName('Settings'); const setData = setSheet.getDataRange().getValues(); const settings = {}; for (let i = 1; i < setData.length; i++) { const [key, value] = setData[i]; if (key) settings[String(key).trim()] = String(value || '').trim(); } const officeDayStr = settings['officeDays'] || '1,2,3,4,5'; const officeDays = officeDayStr.split(',').map(s => parseInt(s.trim())).filter(n => !isNaN(n)); // Closures const clSheet = ss.getSheetByName('Closures'); const closures = []; if (clSheet) { const clData = clSheet.getDataRange().getValues(); for (let i = 1; i < clData.length; i++) { const [date] = clData[i]; if (!date) continue; if (date instanceof Date) { const y = date.getFullYear(); const m = String(date.getMonth() + 1).padStart(2, '0'); const d = String(date.getDate()).padStart(2, '0'); closures.push(`${y}-${m}-${d}`); } else { const s = String(date).trim(); if (s) closures.push(s); } } } return { events, officeDays, closures, logoUrl: settings['logoUrl'] || '' }; }
